cmake_minimum_required(VERSION 3.16)
project(navapex_ros2 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)

find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(nav2_msgs REQUIRED)
find_package(nav2_costmap_2d REQUIRED)
find_package(lifecycle_msgs REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(std_srvs REQUIRED)
find_package(tf2 REQUIRED)
find_package(tf2_ros REQUIRED)
find_package(tf2_geometry_msgs REQUIRED)
find_package(Python3 REQUIRED COMPONENTS Interpreter)

get_filename_component(NAVAPEX_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/.." ABSOLUTE)
add_subdirectory("${NAVAPEX_ROOT}" "${CMAKE_CURRENT_BINARY_DIR}/navapex")
add_executable(navapex_node src/navapex_node.cpp)
target_include_directories(navapex_node PRIVATE include)
target_link_libraries(navapex_node planner::core planner::log)
ament_target_dependencies(navapex_node rclcpp nav_msgs nav2_msgs geometry_msgs std_msgs std_srvs
  tf2 tf2_ros tf2_geometry_msgs)
target_compile_options(navapex_node PRIVATE -Wall -Wextra -Wpedantic)
install(TARGETS navapex_node DESTINATION lib/${PROJECT_NAME})
add_executable(navapex_costmap_node src/costmap_node.cpp)
ament_target_dependencies(navapex_costmap_node rclcpp nav2_costmap_2d lifecycle_msgs)
install(TARGETS navapex_costmap_node DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY launch config worlds rviz DESTINATION share/${PROJECT_NAME})
install(DIRECTORY "${NAVAPEX_ROOT}/planner/path_plan/config/" DESTINATION share/${PROJECT_NAME}/config/path_plan)
install(DIRECTORY "${NAVAPEX_ROOT}/planner/config/" DESTINATION share/${PROJECT_NAME}/config/planner
  FILES_MATCHING PATTERN "*.json")

file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/maps")
configure_file(maps/navapex.yaml "${CMAKE_CURRENT_BINARY_DIR}/maps/navapex.yaml" COPYONLY)
execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/scripts/generate_map.py"
  --world "${CMAKE_CURRENT_SOURCE_DIR}/worlds/navapex.world"
  --output "${CMAKE_CURRENT_BINARY_DIR}/maps/navapex.pgm" COMMAND_ERROR_IS_FATAL ANY)
install(DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/maps/" DESTINATION share/${PROJECT_NAME}/maps)
ament_package()
